// invoice.ID is a UUIDv7 generated by the database; it is unique per invoice.
func (m *Mailer) SendInvoice(ctx context.Context, inv Invoice) error {
	key := "invoice-email:" + inv.ID
	ok, err := m.store.SetNX(ctx, key, 1, 24*time.Hour)
	if err != nil || !ok {
		return err
	}
	return m.smtp.Send(ctx, inv.Email())
}
